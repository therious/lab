rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // All data is scoped under apps/{appId} where appId is derived from
    // the client's hostname+port (e.g. "localhost_5173", "app.example.com").
    // This allows multiple apps to share one Firebase project without collision.
    match /apps/{appId} {

      // ── Users ──────────────────────────────────────────────────────────────
      // Any authenticated user can read all profiles (needed for the users grid).
      // Only the owner can write their own document (presence, lastSeen, etc).
      match /users/{uid} {
        allow read:  if request.auth != null;
        allow write: if request.auth.uid == uid;
      }

      // ── 1-1 chats ──────────────────────────────────────────────────────────
      // chatId is built as [uidA, uidB].sort().join('__').
      // Participants are derived directly from the document ID — no ACL document needed.
      match /chats/{chatId}/messages/{msgId} {
        allow read:   if request.auth != null
                      && request.auth.uid in chatId.split('__');
        allow create: if request.auth != null
                      && request.auth.uid in chatId.split('__')
                      && request.resource.data.from == request.auth.uid;
      }

      // ── Group chats ────────────────────────────────────────────────────────
      // chatId is a Firestore auto-generated ID.
      // Participants are stored as a string array field on the document itself.
      //
      // NOTE: `allow list` is intentionally separated from `allow get`.
      // The client always queries with where('participants','array-contains',uid)
      // so in practice only the user's own groups are returned.  Firestore's
      // rule engine cannot always statically prove that
      //   `request.auth.uid in resource.data.participants`
      // is implied by the array-contains filter, so we allow list for any
      // authenticated user and rely on the query constraint to scope results.
      match /groupChats/{chatId} {
        allow get:    if request.auth != null
                      && request.auth.uid in resource.data.participants;
        allow list:   if request.auth != null;
        allow create: if request.auth != null
                      && request.auth.uid in request.resource.data.participants
                      && request.resource.data.createdBy == request.auth.uid;
        // Allow participants to update lastMessageAt and nickname only
        allow update: if request.auth != null
                      && request.auth.uid in resource.data.participants;
      }

      // Messages subcollection intentionally does NOT use get() to look up
      // participants: get() is unsupported in list (query) rules and would
      // silently deny all snapshot listeners.  Security is provided by the
      // unguessable Firestore-generated chatId — clients only discover group
      // IDs through the scoped groupChats query above.
      match /groupChats/{chatId}/messages/{msgId} {
        allow read:   if request.auth != null;
        allow create: if request.auth != null
                      && request.resource.data.from == request.auth.uid;
      }

    }
  }
}
